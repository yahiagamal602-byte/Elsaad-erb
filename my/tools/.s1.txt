145 | function validateVoucher(database: DatabaseSync, partyKind: FinancialVoucherParty, value: unknown): FinancialVoucherInput {
146 |   if (!value || typeof value !== 'object') throw new Error('بيانات السند غير صالحة.');
147 |   const input = value as Partial<FinancialVoucherInput>;
148 |   if (input.id !== undefined) validId(input.id, 'رقم السند');
149 |   if (input.partyKind !== partyKind) throw new Error('نوع الطرف في السند غير صالح.');
150 |   if (!validDate(input.date)) throw new Error('تاريخ السند غير صالح.');
151 |   const partyId = validId(input.partyId, partyKind === 'customer' ? 'العميل' : 'المورد');
152 |   if (typeof input.amount !== 'string' || parseAmount(input.amount) <= 0n) throw new Error('مبلغ السند يجب أن يكون أكبر من صفر.');
153 |   if (input.paymentMethod !== 'cash' && input.paymentMethod !== 'bank') throw new Error('طريقة الدفع غير صالحة.');
154 |   const cashBoxId = input.cashBoxId === null ? null : validId(input.cashBoxId, 'الخزينة');
155 |   const bankAccountId = input.bankAccountId === null ? null : validId(input.bankAccountId, 'البنك');
156 |   const costCenterId = input.costCenterId === null ? null : validId(input.costCenterId, 'مركز التكلفة');
157 |   const projectId = input.projectId === null ? null : validId(input.projectId, 'المشروع');

160 |   const party = partyKind === 'customer'

177 | function voucherQuery(partyKind: FinancialVoucherParty): string {
178 |   return `SELECT vouchers.*, COALESCE(customers.code, suppliers.code, '') AS party_code,
179 |       COALESCE(customers.name, suppliers.name, '') AS party_name,
180 |       COALESCE(customers.account_id, suppliers.account_id) AS party_account_id,
181 |       COALESCE(boxes.name, banks.name, '') AS payout_name,
182 |       COALESCE(boxes.currency, banks.currency, 'EGP') AS payout_currency
183 |     FROM financial_vouchers vouchers
184 |     LEFT JOIN customers ON customers.id = vouchers.customer_id
185 |     LEFT JOIN suppliers ON suppliers.id = vouchers.supplier_id
186 |     LEFT JOIN treasury_cash_boxes boxes ON boxes.id = vouchers.cash_box_id
187 |     LEFT JOIN treasury_bank_accounts banks ON banks.id = vouchers.bank_account_id
188 |     WHERE vouchers.party_kind = '${partyKind}'`;
189 | }
