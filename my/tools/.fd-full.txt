34 | function validDate(value: unknown): value is string {
35 |   if (typeof value !== 'string') return false;
36 |   const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(value);
37 |   if (!match) return false;
38 |   const [, year, month, day] = match.map(Number);
39 |   const date = new Date(Date.UTC(year, month - 1, day));
40 |   return date.getUTCFullYear() === year && date.getUTCMonth() === month - 1 && date.getUTCDate() === day;

296 | function createJournal(database: DatabaseSync, requestKey: string, date: string, description: string, lines: Array<{ accountId: number; debit: string; credit: string; description: string; costCenterId: number | null }>): number {
297 |   const existing = database.prepare('SELECT id FROM journal_entries WHERE request_key = ?').get(requestKey) as Row | undefined;
298 |   if (existing) return Number(existing.id);
299 |   const debit = lines.reduce((sum, line) => sum + parseAmount(line.debit), 0n);
300 |   const credit = lines.reduce((sum, line) => sum + parseAmount(line.credit), 0n);
301 |   if (debit <= 0n || debit !== credit) throw new Error('الحركة المالية غير متوازنة ولا يمكن ترحيلها.');
302 |   const journal = database.prepare(`INSERT INTO journal_entries (reference, entry_date, description, request_key) VALUES (?, ?, ?, ?)`)
303 |     .run(requestKey, date, description, requestKey);
304 |   const insert = database.prepare(`INSERT INTO journal_lines (entry_id, line_order, account_id, debit, credit, description, cost_center_id)
305 |     VALUES (?, ?, ?, ?, ?, ?, ?)`);
306 |   lines.forEach((line, index) => insert.run(Number(journal.lastInsertRowid), index + 1, line.accountId, line.debit, line.credit, line.description, line.costCenterId));
307 |   return Number(journal.lastInsertRowid);
